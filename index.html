if not found then if exists (select 1 from public.hour_evidence_reviews ev where ev.id = p_evidence_id) then raise exception 'Evidence record has already been reviewed'; else raise exception 'Evidence record not found'; end if; end if; -- Recalculate from ALL evidence records for this student; never let one rejected -- record override another qualifying approved record.
